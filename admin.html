function updatePreview() {
const fileName = document.getElementById('imageFileName').value.trim();
const img = document.getElementById('imgPreview');
const text = document.getElementById('previewText');
if (fileName) {
// Tự động mã hóa khoảng trắng thành %20 để không bị lỗi link ảnh
const safePath = `images/${encodeURI(fileName)}`;
img.src = safePath;
img.style.display = 'block';
text.innerText = `Đường dẫn web: ${safePath}`;
} else {
img.style.display = 'none';
text.innerText = 'Xem trước ảnh Testpoint';
}
}
function addTestpoint() {
const model = document.getElementById('brandModel').value.trim();
const title = document.getElementById('displayName').value.trim();
const fileName = document.getElementById('imageFileName').value.trim();
if (!title || !fileName) {
alert("Vui lòng nhập đầy đủ thông tin tên máy và tên file ảnh!");
return;
}
const fmtErr = validateImageName(fileName);
if (fmtErr) {
if (!confirm(fmtErr + "\n\nVẫn thêm với định dạng này?")) return;
}
// Tự động xử lý đường dẫn an toàn chứa khoảng trắng
const safeImagePath = `images/${encodeURI(fileName)}`;
const newItem = {
id: model || title.toLowerCase().replace(/\s+/g, '_'),
title: title,
image: safeImagePath
};
let db = JSON.parse(localStorage.getItem('testpoint_db') || '[]');
db.push(newItem);
localStorage.setItem('testpoint_db', JSON.stringify(db));
renderDatabase();
// Reset form
document.getElementById('brandModel').value = '';
document.getElementById('displayName').value = '';
document.getElementById('imageFileName').value = '';
updatePreview();
alert("Đã thêm thành công!");
}
const SUPPORTED = {
'image/png': {ext:'PNG', ok:true, note:'Chuẩn — sắc nét cho sơ đồ testpoint'},
'image/jpeg': {ext:'JPG', ok:true, note:'Nhẹ, phù hợp ảnh chụp mạch'},
'image/webp': {ext:'WEBP', ok:true, note:'Nhẹ nhất, khuyến nghị'},
'image/gif': {ext:'GIF', ok:true, note:'OK nhưng ảnh testpoint không cần animation'},
'image/bmp': {ext:'BMP', ok:true, note:'OK nhưng file nặng'},
'image/svg+xml': {ext:'SVG', ok:true, note:'Vector — sắc nét mọi kích thước'},
'image/x-icon': {ext:'ICO', ok:true, note:'OK (icon nhỏ)'},
};
function onFileChosen() {
const inp = document.getElementById('imageFile');
const nameInput = document.getElementById('imageFileName');
const info = document.getElementById('formatInfo');
const badges = document.getElementById('formatBadges');
if (!inp.files || !inp.files[0]) return;
const f = inp.files[0];
// Điền tên file (kể cả khoảng trắng) vào field tên
nameInput.value = f.name;
updatePreview();
renderBadges(f.type, f.name.split('.').pop().toLowerCase(), f.size);
}
function renderBadges(mime, ext, size) {
const badges = document.getElementById('formatBadges');
const info = document.getElementById('formatInfo');
badges.innerHTML = '';
Object.entries(SUPPORTED).forEach(([m, v]) => {
const b = document.createElement('span');
b.style.cssText = 'padding:4px 10px;border-radius:999px;font-size:12px;font-weight:600;background:' +
(m === mime ? 'linear-gradient(135deg,#6366f1,#8b5cf6);color:#fff' : '#21262d;color:#8b949e') +
';border:1px solid ' + (m === mime ? '#8b5cf6' : '#30363d');
b.textContent = v.ext;
badges.appendChild(b);
});
const meta = SUPPORTED[mime];
if (meta) {
info.innerHTML = `<b style="color:#7ee787">✔ Hỗ trợ ${meta.ext}</b> — ${meta.note} — dung lượng ${(size/1024).toFixed(1)} KB`;
} else {
info.innerHTML = `<b style="color:#f85149">✖ Định dạng .${ext} (${mime || 'không rõ'}) có thể không hiển thị trên trình duyệt</b> — khuyến nghị PNG/JPG/WEBP`;
}
}
function validateImageName(name) {
const okExts = ['png','jpg','jpeg','webp','gif','bmp','svg','ico'];
const ext = name.split('.').pop().toLowerCase();
if (!okExts.includes(ext)) {
return `Định dạng .${ext} không nằm trong danh sách khuyến nghị: ${okExts.join(', ')}`;
}
return null;
}
function renderDatabase() {
let db = JSON.parse(localStorage.getItem('testpoint_db') || '[]');
document.getElementById('databaseOutput').value = JSON.stringify(db, null, 2);
}
function copyDatabase() {
const textarea = document.getElementById('databaseOutput');
textarea.select();
document.execCommand('copy');
alert("Đã copy toàn bộ mã JSON!");
}
function clearLocalDB() {
if (confirm("Bạn có chắc muốn xóa bộ nhớ tạm trên trình duyệt này?")) {
localStorage.removeItem('testpoint_db');
renderDatabase();
}
}
